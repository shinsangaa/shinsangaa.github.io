# error_raise.py

# 오류 일부러 발생시키기 - raise문 사용해 오류 강제 발생시킬 수 있다.

# 여러 명이 함께 작업 시, 실수 방지 위해서

# 자식 클래스가 반드시 구현해야 할 기능을 빠뜨리는 실수를 방지

# '이 기능을 아직 구현 안했다' 사실을 바로 알 수 있다

# NotImplementedError :

꼭 작성해야 하는 부분이 구현되지 않았을 경우 일부러 오류를 발생시키기 위해 사용한다.

class Bird :
def fly(self) :
raise NotImplementedError
class Eagle(Bird) :
pass
eagle = Eagle()
eagle.fly()

>> 실행결과 <<
Traceback (most recent call last):
File "C:/doit/error_raise.py", line 15, in <module>
eagle.fly()
File "C:/doit/error_raise.py", line 9, in fly
raise NotImplementedError
NotImplementedError

​

Eagle 클래스는 Bird 클래스를 상속받았다.

Eagle 클래스에서 fly 메서드를 오버라이딩하여 구현하지 않았다.

eagle 객체의 fly 메서드 수행하는 순간 Bird 클래스의 fly 메서드가 수행되어 NotImplementedError 발생한다.

​

NotImplementedError 오류 발생 안하게 하기 위해 Eagle 클래스에 fly 메서드를 구현한다. (메서드 오버라이딩)

class Bird :
def fly(self) :
raise NotImplementedError
class Eagle(Bird) :
def fly(self) :
print('very fast')
eagle = Eagle()
eagle.fly()

>> 실행결과 <<
very fast

예외 처리 기법 사용

class MyError(Exception) :
pass
def say_nick(nick) :
if nick == '바보' :
raise MyError()
print(nick)
try :
say_nick('천사')
say_nick('바보')
except MyError :
print('허용되지 않는 별명입니다.')

>> 실행 결과 <<
천사

오류 메시지를 사용

class MyError(Exception) :
def __str__(self) :
return "허용되지 않는 별명입니다."
def say_nick(nick) :
if nick == '바보' :
raise MyError()
print(nick)
# 예외 처리 기법 사용
try :
say_nick('천사')
say_nick('바보')
except MyError as e :
print(e)

>> 실행 결과 <<
천사
허용되지 않는 별명입니다.

오류 메시지를 사용하고 싶으면 오류 클래스에 __str__ 메서드를 구현해야 한다.
