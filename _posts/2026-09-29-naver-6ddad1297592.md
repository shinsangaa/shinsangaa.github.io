import csv # csv파일 읽기 위해 csv 라이브러리 설치 (파이썬 내장 라이브러리)
total = 0
file_path = 'venv/20260928/data/card.csv'
with open(file_path, mode='r', encoding='utf-8') as file : # 한글 포함된 파일 다룰 땐 인코딩 명시하자
reader = csv.reader(file) # 한 줄씩 읽은 내용을 reader 변수에 저장
next(reader) # 한 줄 건너뜀 -> 첫 줄의 '이용금액' 건너뛰어서 int(row[6]) 부분 ValueError 안 나게 만듦
for row in reader : # 파일 1줄씩 읽기 반복. 한 줄씩 row에 리스트로 저장
#print(row)
#print(row[6])
total = total + int(row[6])
print(f'총 이용금액: {total:,} 원') # 천자리마다 쉼표 찍기

# 실행결과
총 이용금액: 11,419,825원

그러나 row[6]에 부분취소, 전표매입 등 매입상태가 달라서 실제 총 이용금액은 위의 결과가 아니다.

import csv
total = 0
total_oct = 0
total_nov = 0
total_dec = 0
total_delivery = 0
total_taxi = 0
count_delivery = 0
count_taxi = 0
file_path = 'venv/20260928/data/card.csv'
with open(file_path, mode='r', encoding='utf-8') as file :
reader = csv.reader(file)
next(reader)
for row in reader :
if row[8] == '전표매입' :
total = total + int(row[6])
if '2019-10-' in row[0] :
total_oct = total_oct + int(row[6])
elif '2019-11-' in row[0] :
total_nov = total_nov + int(row[6])
elif '2019-12-' in row[0] :
total_dec = total_dec + int(row[6])
if '택시' in row[5] :
count_taxi += 1
total_taxi = total_taxi + int(row[6])
if row[5] == '(주)우아한형제들' :
count_delivery += 1
total_delivery = total_delivery + int(row[6])
print(f'총 이용금액: {total:,}원')
print(f'10월 평균 이용금액: {total_oct/31:,.1f}원')
print(f'11월 평균 이용금액: {total_nov/30:,.1f}원')
print(f'12월 평균 이용금액: {total_dec/31:,.1f}원')
print(f'택시비 평균 이용금액: {total_taxi/count_taxi:,.1f}원')
print(f'배달음식 평균 이용금액: {total_delivery/count_delivery:,.1f}원')
