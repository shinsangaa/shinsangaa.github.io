from openai import OpenAI
from dotenv import load_dotenv # .env 찾기
import os # file이나 folder 담기 위해 필수. .env파일의 위치를 찾음
load_dotenv() # .env파일 실행
api_key = os.getenv('OPENAI_API_KEY')
client = OpenAI(api_key = api_key) # api_key를 OpenAI서버로 보내 인증 후 다시 돌아와 client에 넣어줌
# key값을 갖고 있는 client / 답변내용을 response에 저장
response = client.chat.completions.create(
model = "gpt-4o",
temperature = 0.9, # 반드시 소수점 작성 0.0, 1.0 등
messages = [
{"role" : "system", "content" : "너는 백설공주 이야기 속의 거울이야. 그 이야기 속의 마법 거울의 캐릭터에 부합하게 답변해줘."},
{"role" : "user", "content" : "세상에서 누가 제일 아름답니?"},
]
)
print(response)
print('----')

멀티턴 대화하는 챗봇 만들기

1. **GPT가 과거 대화 내용을 기억하지 못함**

사용자는 터미널에 메시지 입력 가능, exit 입력할 때 까지 대화 지속된다

single_turn.py

from openai import OpenAI
from dotenv import load_dotenv
import os
load_dotenv()
api_key = os.getenv('OPENAI_API_KEY')
client = OpenAI(api_key = api_key)
while True : # while 뒤 조건문 작성. True 작성 시 무한루프
user_input = input("사용자: ")
if user_input == "exit": # if문 뒤 exit 입력 시 break로 루프 빠져나옴
break
response = client.chat.completions.create(
model = "gpt-4o",
temperature = 0.9,
messages = [
{"role": "system", "content": "너는 사용자를 도와주는 상담사야."},
{"role": "user", "content": user_input},
],
)
print("AI: " + response.choices[0].message.content)

2. 사용자의 입력값과 GPT가 생성한 답변을 messages에 추가하도록 만들기 (Multi-Turn)

multi_turn.py

from openai import OpenAI
from dotenv import load_dotenv
import os
load_dotenv()
api_key = os.getenv('OPENAI_API_KEY')
client = OpenAI(api_key = api_key)
def get_ai_response(message):
response = client.chat.completions.create(
model = "gpt-4o",
temperature = 0.9,
messages = messages # 대화 기록을 입력으로 전달
)
return response.choices[0].message.content # 생성된 응답의 내용 반환
messages = [
{"role": "system", "content": "너는 사용자를 도와주는 상담사야."}, # 초기 시스템 메시지
]
while True :
user_input = input("사용자: ") # 사용자 입력 받기
if user_input == "exit": # 사용자가 대화를 종료하려는지 확인
break
messages.append({"role": "user", "content": user_input}) # 사용자 메시지를 대화기록에 추가
ai_response = get_ai_response(messages) # 대화 기록을 기반으로 AI 응답 가져오기
messages.append({"role": "assistant", "content": ai_response}) # AI 응답 대화 기록에 추가하기
print("AI: " + ai_response) # AI 응답 출력

GPT는 항상 입력된 텍스트를 바탕으로 다음에 나올 문장을 예측할 뿐, 과거의 기억을 갖고 있지 않다.

​

멀티턴 소스코드를 보면, messages 변수에 내 질문이 계속 쌓인다.

용량도 커지고, 코드가 무거워진다.

=> 서버에 이전 질문 저장이 필요하다.

​

[![]({{ '/assets/images/naver-import/d47c49e54096/img-010.png' | relative_url }})](#)

OpenAI에 질문했을 때 이동 경로

client가 질문하면, 국내 Server에 접속한다. 질문내용은 Session에 저장된다.

국내의 WAS에서 OpenAI 데이터센터로 이동해 답변을 얻고, 다시 Server로 돌아왔다가 User에게 답변 준다.

client가 두 번째 질문을 하면, 국내 Server에 접속하고, session에 저장된다.

Server는 session에 저장된 것 통채로 OpenAI 데이터 센터로 보낸다.

OpenAI Data Center는 session을 통채로 받았기 때문에 이전 질문을 알 수 있다.
