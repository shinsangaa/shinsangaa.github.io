import csv
file_path = 'venv/20260928/data/card.csv' # 파일 경로
# VScode에서 특히 오류 많이 발생 (실행 위치가 ?)
total_amount = 0
# BOM(\ufeff) 정리를 위해 encoding = 'utf-8-sig' 사용
with open(file_path, mode='r', encoding='utf-8-sig') as file : # open함수는 해당 파일을 열어만 놓은 상태
reader = csv.reader(file) # reader함수 특징 - 엔터 치면 각 줄을 하나로(리스트) 인식
# 1. 헤더(첫번째 줄 스킵)
header = next(reader)
print(header) # 내가 제거한 column이 무엇인지 확인하고 싶을 때. 제거한 열이 필요하다면 header 변수에 저장해놓기
# 2. 각 행을 돌며 '이용금액'(인덱스 6) 합산
for row in reader :
if row : # 빈 줄 방지 (한 줄이 데이터 갖고 있으면 True, 비어 있으면 False) 신입사원이 이런 코드를 잘 넣지 않음!
total_amount += int(row[6])
print(f'총 이용금액: {total_amount:,}원')
